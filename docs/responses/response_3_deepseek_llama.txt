/// Game state enumeration
enum GameState { startScreen, playing, gameOver }

/// Core game logic and physics engine
class GameEngine extends ChangeNotifier {
  static const double gravity = 1200.0; // pixels per second squared
  static const double flapForce = -350.0; // pixels per second
  static const double maxVelocity = 500.0;
  
  // Bird properties
  double birdY = 0;
  double birdVelocity = 0;
  static const double birdSize = 40.0;
  double birdRotation = 0;
  
  // Pipe properties
  List<Pipe> pipes = [];
  static const double pipeWidth = 70.0;
  static const double pipeGap = 180.0;
  static const double pipeSpacing = 300.0;
  double pipeSpawnTimer = 0;
  
  // Game state
  GameState gameState = GameState.startScreen;
  int score = 0;
  int bestScore = 0;
  
  // World dimensions (set when game starts)
  double screenHeight = 0;
  double screenWidth = 0;
  double groundY = 0;
  
  // Visual effects
  double backgroundOffset = 0;
  
  void startGame() {
    gameState = GameState.playing;
    resetGameState();
    notifyListeners();
  }
  
  void resetGame() {
    gameState = GameState.startScreen;
    resetGameState();
    notifyListeners();
  }
  
  void resetGameState() {
    birdY = screenHeight / 2;
    birdVelocity = 0;
    birdRotation = 0;
    pipes.clear();
    pipeSpawnTimer = 0;
    score = 0;
  }
  
  void flap() {
    if (gameState == GameState.playing) {
      birdVelocity = flapForce;
      // Small rotation up on flap
      birdRotation = -0.3;
    }
  }
  
  void update() {
    if (gameState != GameState.playing) return;
    
    // Update bird physics
    final dt = 1 / 60; // Fixed timestep
    birdVelocity += gravity * dt;
    birdVelocity = birdVelocity.clamp(-maxVelocity, maxVelocity);
    birdY += birdVelocity * dt;
    
    // Update bird rotation (smooth transition)
    birdRotation = (birdRotation * 0.9 + (birdVelocity / 500) * 0.1).clamp(-1.2, 1.2);
    
    // Update background
    backgroundOffset = (backgroundOffset + 100 * dt) % 400;
    
    // Spawn pipes
    pipeSpawnTimer += dt;
    if (pipeSpawnTimer >= 1.8) {
      pipeSpawnTimer = 0;
      spawnPipe();
    }
    
    // Update pipes
    for (var i = pipes.length - 1; i >= 0; i--) {
      pipes[i].x -= 150 * dt; // Move left
      if (pipes[i].x + pipeWidth < 0) {
        pipes.removeAt(i);
      }
    }
    
    // Check scoring
    for (var pipe in pipes) {
      if (!pipe.passed && pipe.x + pipeWidth < 100) {
        pipe.passed = true;
        score++;
        if (score > bestScore) {
          bestScore = score;
        }
        // Play score sound here
        notifyListeners();
      }
    }
    
    // Check collisions
    if (checkCollisions()) {
      gameState = GameState.gameOver;
      notifyListeners();
    }
    
    notifyListeners();
  }
  
  void spawnPipe() {
    final random = Random();
    final minGapY = pipeGap / 2 + 50;
    final maxGapY = screenHeight - groundY - pipeGap / 2 - 50;
    final gapY = minGapY + random.nextDouble() * (maxGapY - minGapY);
    
    pipes.add(Pipe(
      x: screenWidth,
      gapY: gapY,
      passed: false,
    ));
  }
  
  bool checkCollisions() {
    // Ground and ceiling collision
    if (birdY <= 0 || birdY + birdSize >= screenHeight - groundY) {
      return true;
    }
    
    // Pipe collision
    final birdRect = Rect.fromLTWH(100, birdY, birdSize, birdSize);
    
    for (var pipe in pipes) {
      final topPipeRect = Rect.fromLTWH(
        pipe.x, 
        0, 
        pipeWidth, 
        pipe.gapY - pipeGap / 2,
      );
      
      final bottomPipeRect = Rect.fromLTWH(
        pipe.x,
        pipe.gapY + pipeGap / 2,
        pipeWidth,
        screenHeight - groundY - (pipe.gapY + pipeGap / 2),
      );
      
      if (birdRect.overlaps(topPipeRect) || birdRect.overlaps(bottomPipeRect)) {
        return true;
      }
    }
    
    return false;
  }
  
  void updateDimensions(double width, double height) {
    screenWidth = width;
    screenHeight = height;
    groundY = height * 0.1; // 10% ground at bottom
    
    if (birdY == 0) {
      birdY = height / 2;
    }
  }
}

/// Pipe data structure
class Pipe {
  double x;
  double gapY;
  bool passed;
  
  Pipe({
    required this.x,
    required this.gapY,
    required this.passed,
  });
}

/// Custom painter for rendering the game
class GamePainter extends CustomPainter {
  final GameEngine gameEngine;
  
  GamePainter({required this.gameEngine});
  
  @override
  void paint(Canvas canvas, Size size) {
    gameEngine.updateDimensions(size.width, size.height);
    
    // Draw sky gradient
    final skyPaint = Paint()..shader = const LinearGradient(
      begin: Alignment.topCenter,
      end: Alignment.bottomCenter,
      colors: [Color(0xFF87CEEB), Color(0xFF98D8E8)],
    ).createShader(Rect.fromLTWH(0, 0, size.width, size.height));
    canvas.drawRect(Rect.fromLTWH(0, 0, size.width, size.height), skyPaint);
    
    // Draw clouds (scrolling)
    _drawClouds(canvas, size);
    
    // Draw pipes
    _drawPipes(canvas, size);
    
    // Draw bird
    _drawBird(canvas, size);
    
    // Draw ground
    _drawGround(canvas, size);
  }
  
  void _drawClouds(Canvas canvas, Size size) {
    final cloudPaint = Paint()..color = Colors.white.withOpacity(0.7);
    
    for (int i = 0; i < 5; i++) {
      final cloudX = (i * 200 + gameEngine.backgroundOffset) % (size.width + 200) - 100;
      final cloudY = 50 + (i * 80) % 150;
      
      canvas.drawCircle(Offset(cloudX, cloudY), 30, cloudPaint);
      canvas.drawCircle(Offset(cloudX + 30, cloudY - 15), 35, cloudPaint);
      canvas.drawCircle(Offset(cloudX - 20, cloudY - 10), 25, cloudPaint);
      canvas.drawCircle(Offset(cloudX + 10, cloudY + 10), 28, cloudPaint);
    }
  }
  
  void _drawPipes(Canvas canvas, Size size) {
    final pipePaint = Paint()
      ..shader = const LinearGradient(
        colors: [Color(0xFF2E7D32), Color(0xFF1B5E20)],
      ).createShader(Rect.fromLTWH(0, 0, 70, 500));
    
    final pipeCapPaint = Paint()..color = const Color(0xFF1B5E20);
    
    for (var pipe in gameEngine.pipes) {
      final topPipeHeight = pipe.gapY - GameEngine.pipeGap / 2;
      final bottomPipeY = pipe.gapY + GameEngine.pipeGap / 2;
      final bottomPipeHeight = size.height - gameEngine.groundY - bottomPipeY;
      
      // Top pipe
      canvas.drawRect(
        Rect.fromLTWH(pipe.x, 0, GameEngine.pipeWidth, topPipeHeight),
        pipePaint,
      );
      // Top pipe cap
      canvas.drawRect(
        Rect.fromLTWH(pipe.x - 10, topPipeHeight - 40, GameEngine.pipeWidth + 20, 40),
        pipeCapPaint,
      );
      
      // Bottom pipe
      canvas.drawRect(
        Rect.fromLTWH(pipe.x, bottomPipeY, GameEngine.pipeWidth, bottomPipeHeight),
        pipePaint,
      );
      // Bottom pipe cap
      canvas.drawRect(
        Rect.fromLTWH(pipe.x - 10, bottomPipeY, GameEngine.pipeWidth + 20, 40),
        pipeCapPaint,
      );
    }
  }
  
  void _drawBird(Canvas canvas, Size size) {
    final birdX = 100.0;
    final birdY = gameEngine.birdY;
    final birdSize = GameEngine.birdSize;
    
    canvas.save();
    canvas.translate(birdX + birdSize / 2, birdY + birdSize / 2);
    canvas.rotate(gameEngine.birdRotation);
    
    // Bird body (gradient)
    final bodyPaint = Paint()
      ..shader = const RadialGradient(
        colors: [Color(0xFFFFD54F), Color(0xFFFFA000)],
      ).createShader(Rect.fromLTWH(-birdSize/2, -birdSize/2, birdSize, birdSize));
    
    canvas.drawRRect(
      RRect.fromRectAndRadius(
        Rect.fromLTWH(-birdSize/2, -birdSize/2, birdSize, birdSize),
        const Radius.circular(20),
      ),
      bodyPaint,
    );
    
    // Eye
    final eyePaint = Paint()..color = Colors.white;
    canvas.drawCircle(const Offset(10, -10), 6, eyePaint);
    canvas.drawCircle(const Offset(12, -10), 3, Paint()..color = Colors.black);
    
    // Beak
    final beakPaint = Paint()..color = const Color(0xFFFF6F00);
    canvas.drawRRect(
      RRect.fromRectAndRadius(
        Rect.fromLTWH(15, -5, 15, 10),
        const Radius.circular(3),
      ),
      beakPaint,
    );
    
    // Wing
    final wingPaint = Paint()..color = const Color(0xFFF57C00);
    canvas.drawRRect(
      RRect.fromRectAndRadius(
        Rect.fromLTWH(-15, -5, 20, 25),
        const Radius.circular(10),
      ),
      wingPaint,
    );
    
    canvas.restore();
  }
  
  void _drawGround(Canvas canvas, Size size) {
    final groundPaint = Paint()..color = const Color(0xFF8D6E63);
    final grassPaint = Paint()..color = const Color(0xFF4CAF50);
    
    final groundHeight = gameEngine.groundY;
    final groundY = size.height - groundHeight;
    
    // Ground base
    canvas.drawRect(
      Rect.fromLTWH(0, groundY, size.width, groundHeight),
      groundPaint,
    );
    
    // Grass top
    canvas.drawRect(
      Rect.fromLTWH(0, groundY - 8, size.width, 8),
      grassPaint,
    );
    
    // Ground texture (dots)
    final dotPaint = Paint()..color = Colors.brown.shade800;
    for (double x = 0; x < size.width; x += 30) {
      for (double y = groundY + 10; y < groundY + groundHeight; y += 20) {
        canvas.drawCircle(Offset(x, y), 3, dotPaint);
      }
    }
  }
  
  @override
  bool shouldRepaint(GamePainter oldDelegate) => true;
}
